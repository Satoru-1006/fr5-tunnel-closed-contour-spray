cmake_minimum_required(VERSION 3.16)
project(stage23a5_backend_equivalent)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FCL REQUIRED fcl)
find_package(Bullet REQUIRED)
add_executable(stage23a5_geometry_probe stage23a5_geometry_probe.cpp)
target_include_directories(stage23a5_geometry_probe PRIVATE ${FCL_INCLUDE_DIRS} ${BULLET_INCLUDE_DIRS})
target_link_directories(stage23a5_geometry_probe PRIVATE ${FCL_LIBRARY_DIRS} ${BULLET_LIBRARY_DIRS})
target_link_libraries(stage23a5_geometry_probe PRIVATE ${FCL_LIBRARIES} BulletCollision BulletDynamics LinearMath)
# The distro pkg-config file advertises -std=c++11; the probe uses
# std::filesystem and is compiled under the package-wide C++17 setting.
find_package(geometry_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(shape_msgs REQUIRED)
add_executable(stage23a5_native_probe stage23a5_native_probe.cpp)
ament_target_dependencies(stage23a5_native_probe geometry_msgs moveit_core moveit_ros_planning moveit_msgs rclcpp shape_msgs)
target_link_libraries(stage23a5_native_probe Eigen3::Eigen)
install(TARGETS stage23a5_geometry_probe stage23a5_native_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
