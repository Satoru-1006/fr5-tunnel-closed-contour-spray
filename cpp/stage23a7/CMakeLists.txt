cmake_minimum_required(VERSION 3.16)
project(stage23a7_runtime_audit)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(shape_msgs REQUIRED)
add_executable(stage23a7_runtime_audit_probe stage23a7_runtime_audit_probe.cpp)
add_executable(stage23a7_runtime_audit_callback_probe stage23a7_runtime_audit_callback_probe.cpp)
add_executable(stage23a7_perturbation_probe stage23a7_perturbation_probe.cpp)
add_executable(stage23a7_2_bullet_exact_replay stage23a7_2_bullet_exact_replay.cpp)
ament_target_dependencies(stage23a7_runtime_audit_probe geometry_msgs moveit_core moveit_ros_planning moveit_msgs rclcpp shape_msgs)
ament_target_dependencies(stage23a7_runtime_audit_callback_probe geometry_msgs moveit_core moveit_ros_planning moveit_msgs rclcpp shape_msgs)
ament_target_dependencies(stage23a7_perturbation_probe geometry_msgs moveit_core moveit_ros_planning moveit_msgs rclcpp shape_msgs)
target_link_libraries(stage23a7_runtime_audit_probe Eigen3::Eigen dl)
target_link_libraries(stage23a7_runtime_audit_callback_probe Eigen3::Eigen dl crypto)
target_link_libraries(stage23a7_perturbation_probe Eigen3::Eigen)
target_include_directories(stage23a7_2_bullet_exact_replay PRIVATE /usr/include/bullet)
target_link_libraries(stage23a7_2_bullet_exact_replay BulletCollision LinearMath)
install(TARGETS stage23a7_runtime_audit_probe stage23a7_runtime_audit_callback_probe stage23a7_perturbation_probe stage23a7_2_bullet_exact_replay DESTINATION lib/${PROJECT_NAME})
ament_package()
