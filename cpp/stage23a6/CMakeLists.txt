cmake_minimum_required(VERSION 3.16)
project(stage23a6_self_collision_audit)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(shape_msgs REQUIRED)
add_executable(stage23a6_audit_probe stage23a6_audit_probe.cpp)
ament_target_dependencies(stage23a6_audit_probe geometry_msgs moveit_core moveit_ros_planning moveit_msgs rclcpp shape_msgs)
target_link_libraries(stage23a6_audit_probe Eigen3::Eigen)
install(TARGETS stage23a6_audit_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
